# R-type (no shift):
ADD   R1  R2  R3
SUB   R3  R4  R5
MUL   R4  R5  R6
AND   R7  R8  R9

# I-type:
MOVI  R1      100
JEQ   R1  R2 -2
XORI  R3  R4  7
MOVR  R5  R6  12   # load R5 from MEM[R6+12]
MOVM  R7  R8  16   # store R7 into MEM[R8+16]

# R-shifts:
LSL   R1  R2   3
LSR   R3  R4   1

# J-type:
JMP   20         # jump to address
